# RFB.txt
[ppl.export-home-variable] ZQB03
// a function called F, one argument
EXPORT F(za)
BEGIN
  RETURN 5020 + za;
END;

EXPORT ZRFB()
BEGIN
  RETURN F(1);
END;

# RLA.txt
[ppl.local-block-scope] ZQB04
// IF: only assigned after
EXPORT ZRLA()
BEGIN
  IF 1 == 1 THEN
    LOCAL zb;
    zb := 3;
  END;
  zb := 4;
  RETURN 5170;
END;

# RLE.txt
[ppl.local-block-scope] ZQB05
// ELSE: read after
EXPORT ZRLE()
BEGIN
  IF 0 == 1 THEN
    1;
  ELSE
    LOCAL zb;
    zb := 3;
  END;
  RETURN zb + 5140;
END;

# RLT.txt

  END;

END;

# RLF.txt

END;

# RLFP.txt
[ppl.local-block-scope] ZQB08
// FOR: read inside, its pair
EXPORT ZRLFP()
BEGIN
  LOCAL zi, zs;
  zs := 0;
  FOR zi FROM 1 TO 2 DO
    LOCAL zb;
    zb := zi;
    zs := zs + zb;
  END;
  RETURN zs + 5210;
END;
