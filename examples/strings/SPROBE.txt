// ===================================================================
// SPROBE - what do the string functions actually do?
//
// The interpreter deliberately does NOT implement LEFT, RIGHT, MID,
// INSTRING or SORT, because their edge behaviour is not measured, and a
// guessed semantics would return a number where the calculator returns
// another -- exactly the divergence the interpreter exists to catch.
//
// This measures them. Run SPROBE on Home (no parentheses), read the
// screen, and report what it says.
//
// If it does not compile, tell me the line: it means one of these
// commands does not take the arguments used here, which is itself the
// answer to part of the question.
// ===================================================================

// Drain what is pending, then wait for a real key. Without this the program
// returns, Home redraws over the screen, and all you see is the result.
EXPORT SPW()
BEGIN
  LOCAL zk;
  REPEAT zk := GETKEY; UNTIL zk < 0;
  REPEAT zk := GETKEY; UNTIL zk >= 0;
  RETURN zk;
END;

// A string result, in brackets so an empty one is visible.
EXPORT SPS(zn, zv, zy)
BEGIN
  TEXTOUT_P(zn + " = [" + zv + "]", 2, zy, 2, RGB(0,0,0), 316);
  RETURN zy + 17;
END;

// A numeric result.
EXPORT SPN(zn, zv, zy)
BEGIN
  TEXTOUT_P(zn + " = " + STRING(zv), 2, zy, 2, RGB(0,0,0), 316);
  RETURN zy + 17;
END;

EXPORT SPROBE()
BEGIN
  LOCAL zy, zs;
  zs := "abcdef";
  RECT();
  TEXTOUT_P("s = abcdef", 2, 2, 3, RGB(0,0,0), 316);
  zy := 26;
  zy := SPS("LEFT(s,3)", LEFT(zs, 3), zy);
  zy := SPS("RIGHT(s,3)", RIGHT(zs, 3), zy);
  zy := SPS("MID(s,2,3)", MID(zs, 2, 3), zy);
  zy := SPN("INSTRING(s,cd)", INSTRING(zs, "cd"), zy);
  zy := SPN("INSTRING(s,zz)", INSTRING(zs, "zz"), zy);
  zy := SPN("INSTRING(s,a)", INSTRING(zs, "a"), zy);
  zy := SPS("LEFT(s,0)", LEFT(zs, 0), zy);
  zy := SPS("LEFT(s,99)", LEFT(zs, 99), zy);
  zy := SPN("SIZE(LEFT(s,99))", SIZE(LEFT(zs, 99)), zy);
  zy := SPS("MID(s,4,99)", MID(zs, 4, 99), zy);
  TEXTOUT_P("press a key to leave", 2, 206, 2, RGB(0,0,0), 316);
  SPW();
  RETURN 1;
END;
