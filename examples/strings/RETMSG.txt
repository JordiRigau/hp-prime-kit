// ===================================================================
// RETMSG - what is the box Home shows when a program ends?
//
// Running a program from Home leaves a small framed line behind, with
// the program's name and a number:  SPROBE2 1
//
// The hypothesis worth testing is that it is nothing special: it is the
// value the program returned, shown the way Home shows the value of
// anything you type. If that is right, a program that returns nothing
// leaves nothing behind.
//
// Run these two from Home, one at a time, no parentheses, and note what
// appears after each:
//
//   RMVAL    returns 42       -> expect a box saying 42
//   RMNONE   returns nothing  -> if the hypothesis holds, no box at all
//
// Both wait for a key first, so you can see the screen before the
// program ends.
// ===================================================================

EXPORT RMW()
BEGIN
  LOCAL zk;
  REPEAT zk := GETKEY; UNTIL zk < 0;
  REPEAT zk := GETKEY; UNTIL zk >= 0;
  RETURN zk;
END;

EXPORT RMVAL()
BEGIN
  RECT();
  TEXTOUT_P("RMVAL: this one returns 42", 2, 40, 3, RGB(0,0,0), 316);
  TEXTOUT_P("press a key, then look at Home", 2, 80, 2, RGB(0,0,0), 316);
  RMW();
  RETURN 42;
END;

EXPORT RMNONE()
BEGIN
  RECT();
  TEXTOUT_P("RMNONE: this one returns nothing", 2, 40, 3, RGB(0,0,0), 316);
  TEXTOUT_P("press a key, then look at Home", 2, 80, 2, RGB(0,0,0), 316);
  RMW();
END;
