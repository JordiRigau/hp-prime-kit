// ===================================================================
// MID2 - does MID take two arguments?
//
// Alone in its own file on purpose. If MID(s, start) is not a valid
// signature the calculator refuses to COMPILE, and a compile error
// takes the whole file with it -- so it must not share a file with
// cases that do work.
//
// Run MID2 on Home (no parentheses). Three outcomes:
//   a string  -> the two-argument form exists, and that is what it gives
//   [ERROR]   -> it compiles, and fails at run time
//   it will not compile -> the form does not exist. Tell me the message.
// ===================================================================

EXPORT MID2()
BEGIN
  LOCAL zr, zk;
  IFERR zr := MID("abcdef", 2); THEN zr := "ERROR"; END;
  RECT();
  TEXTOUT_P("MID(abcdef,2) = [" + zr + "]", 2, 40, 3, RGB(0,0,0), 316);
  TEXTOUT_P("press a key to leave", 2, 206, 2, RGB(0,0,0), 316);
  REPEAT zk := GETKEY; UNTIL zk < 0;
  REPEAT zk := GETKEY; UNTIL zk >= 0;
  RETURN 1;
END;
