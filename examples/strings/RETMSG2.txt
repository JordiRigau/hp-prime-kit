// ===================================================================
// RETMSG2 - can a program end without leaving a line on Home?
//
// A function with no RETURN answers with its last bare expression --
// measured. So the question left is what happens when the last
// statement is NOT an expression. Four endings, one per function:
//
//   RM2ASG   ends in an assignment
//   RM2LOOP  ends in a FOR
//   RM2IF    ends in an IF whose condition is false
//   RM2CALL  ends in a call to a function that itself ends in an
//            assignment
//
// Run each on Home, no parentheses, and note what the line says --
// or whether there is no line at all, which is the answer we are
// after. The PC interpreter answers 0 for all four; if the calculator
// answers something else, the rule reaches deeper than that.
//
// THE WAIT COMES BEFORE THE LAST STATEMENT ON PURPOSE. If it came
// last, the key code would BE the last expression, and every function
// would answer with the key you pressed -- measuring the probe
// instead of the question.
// ===================================================================

EXPORT RM2W()
BEGIN
  LOCAL zk;
  REPEAT zk := GETKEY; UNTIL zk < 0;
  REPEAT zk := GETKEY; UNTIL zk >= 0;
  RETURN zk;
END;

EXPORT RM2SHOW(zt)
BEGIN
  RECT();
  TEXTOUT_P(zt, 2, 40, 3, RGB(0,0,0), 316);
  TEXTOUT_P("press a key, then look at Home", 2, 80, 2, RGB(0,0,0), 316);
  RM2W();
  RETURN 0;
END;

EXPORT RM2ASG()
BEGIN
  LOCAL z;
  RM2SHOW("RM2ASG: ends in an assignment");
  z := 1;
END;

EXPORT RM2LOOP()
BEGIN
  LOCAL z, zi;
  RM2SHOW("RM2LOOP: ends in a FOR");
  z := 0;
  FOR zi FROM 1 TO 2 DO
    z := zi;
  END;
END;

EXPORT RM2IF()
BEGIN
  LOCAL z;
  RM2SHOW("RM2IF: ends in an IF that does not run");
  z := 0;
  IF z == 1 THEN
    z := 2;
  END;
END;

EXPORT RM2CALL()
BEGIN
  RM2SHOW("RM2CALL: ends in a call to RM2ASG");
  RM2ASG();
END;
