// PROVAESC - prova de l'escriptor de .hpprgm
//
// Aquest fitxer NO s'ha escrit mai des de la calculadora ni des del
// Connectivity Kit: l'ha generat scripts/hpprgm.py des de Python, agafant
// la capcalera d'una plantilla i posant-hi aquest text a dins.
//
// Si el veus aqui, amb els accents bé (à, è, ó, ç) i les dues funcions
// donant el que toca, l'escriptor funciona.
// ===================================================================

// Ha de tornar 385 (la suma dels quadrats d'1 a 10).
EXPORT PESC1()
BEGIN
  LOCAL zi, zs;
  zs := 0;
  FOR zi FROM 1 TO 10 DO
    zs := zs + zi * zi;
  END;
  RETURN zs;
END;

// Ha de tornar la cadena sencera, amb accents.
EXPORT PESC2()
BEGIN
  RETURN "escriptor OK - àèóç - 4242";
END;

// Ha de tornar 1 si les dues anteriors donen el que toca, 0 si no.
EXPORT PESC3()
BEGIN
  IF PESC1() == 385 AND SIZE(PESC2()) == 26 THEN
    RETURN 1;
  END;
  RETURN 0;
END;
